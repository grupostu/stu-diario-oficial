            </h4>
            <p class="text-xs text-gs-text2 leading-relaxed">

function evaluarCriterio(entrada) {
  var texto = String(entrada == null ? '' : entrada).replace(/\s+/g, ' ').trim();
  var fallo = function (codigo, error) { return { ok: false, codigo: codigo, error: error }; };
  if (texto.length < 3) return fallo('corto', 'Escribe al menos 3 caracteres del DUI o nombre a vigilar.');
  if (texto.length > 120) return fallo('largo', 'El criterio admite 120 caracteres como máximo.');
  if (/[<>]/.test(texto)) return fallo('etiquetas', 'No uses etiquetas ni los signos < >. Escribe solo letras, números y guiones.');
  var digitos = texto.replace(/\D/g, '');
  var letras = (texto.match(/\p{L}/gu) || []).length;
  if (letras === 0 && digitos.length === 0) return fallo('solo_simbolos', 'Escribe letras o números: con solo símbolos no se puede buscar ni vigilar.');
  if (!/^[\p{L}\p{M}0-9 .,'’–-]+$/u.test(texto)) {
    return fallo('simbolos', 'Usa solo letras, números, espacios y guiones; quita los símbolos.');
  }
  if (letras === 0) {
    if (digitos.length < 9) return fallo('dui_incompleto', 'El DUI está incompleto: lleva 9 dígitos (8 más el verificador), por ejemplo 00000000-0.');
    if (digitos.length > 9) return fallo('dui_largo', 'El DUI tiene demasiados dígitos: son 9 (8 más el verificador), por ejemplo 00000000-0.');
    var suma = 0;
    for (var i = 0; i < 8; i++) suma += Number(digitos.charAt(i)) * (9 - i);
    if ((10 - (suma % 10)) % 10 !== Number(digitos.charAt(8))) {
      return fallo('dui_verificador', 'El dígito verificador de este DUI no es válido. Revisa que lo hayas escrito bien.');
    }
    return { ok: true, valor: digitos.slice(0, 8) + '-' + digitos.slice(8), tipo: 'DUI' };
  }
  if (digitos.length > 0 && /^[\p{L}0-9]{8}\s?[–-]\s?[\p{L}0-9]$/u.test(texto)) {
    return fallo('dui_forma', 'Un DUI solo lleva números: 8 dígitos, un guion y el dígito verificador, por ejemplo 00000000-0.');
  }
  if (letras < 3) return fallo('corto', 'Un nombre necesita al menos 3 letras.');
  return { ok: true, valor: texto, tipo: 'NOMBRE' };
}

/**
 * Pliega el texto: minúsculas, sin diacríticos y con cada racha de espacios reducida a uno.
 * Devuelve el texto plegado y, por cada unidad UTF-16 plegada, el rango [ini, fin) del original.
 */
function gsPlegar(texto) {
  texto = String(texto == null ? '' : texto);
  var out = [], ini = [], fin = [];
  var espera = false, eIni = 0, eFin = 0;
  function emitir(cadena, a, b) {
    for (var u = 0; u < cadena.length; u++) { out.push(cadena.charAt(u)); ini.push(a); fin.push(b); }
  }
  var i = 0;
  while (i < texto.length) {
    var code = texto.charCodeAt(i);
    var largo = 1;
    if (code >= 0xD800 && code <= 0xDBFF && i + 1 < texto.length) {
      var lo = texto.charCodeAt(i + 1);
      if (lo >= 0xDC00 && lo <= 0xDFFF) largo = 2;
    }
    var fin_i = i + largo;
    // Camino rápido ASCII
    if (code < 128) {
      if (code === 32 || (code >= 9 && code <= 13)) {
        if (!espera) { espera = true; eIni = i; }
        eFin = fin_i;
      } else {
        if (espera) { if (out.length > 0) emitir(' ', eIni, eFin); espera = false; }
        var c = code >= 65 && code <= 90 ? code + 32 : code;
        emitir(String.fromCharCode(c), i, fin_i);
      }
      i = fin_i;
      continue;
    }
    var ch = texto.slice(i, fin_i);
    if (/\s/.test(ch)) {
      if (!espera) { espera = true; eIni = i; }
      eFin = fin_i;
    } else if (code === 0xAD || (code >= 0x200B && code <= 0x200F) || code === 0xFEFF || code === 0x2060) {
      // Invisibles (guion blando, ancho cero): se ignoran pero se absorben en el rango previo
      if (fin.length > 0 && !espera) fin[fin.length - 1] = fin_i;
    } else {
      if (espera) { if (out.length > 0) emitir(' ', eIni, eFin); espera = false; }
      var partes = Array.from(ch.normalize('NFKD'));
      for (var k = 0; k < partes.length; k++) {
        var p = partes[k];
        if (/\p{M}/u.test(p)) { if (fin.length > 0) fin[fin.length - 1] = fin_i; continue; }
        if (p === '’' || p === '‘' || p === '‛' || p === '´') p = "'";
        else if (p === '“' || p === '”' || p === '„' || p === '«' || p === '»') p = '"';
        else if (p === '–' || p === '—' || p === '−' || p === '‐' || p === '‑') p = '-';
        var bajo = Array.from(p.toLowerCase().normalize('NFKD'));
        for (var m = 0; m < bajo.length; m++) {
          if (/\p{M}/u.test(bajo[m])) continue;
          emitir(GS_ESPECIALES[bajo[m]] || bajo[m], i, fin_i);
        }
      }
    }
    i = fin_i;
  }
  return { plano: out.join(''), ini: ini, fin: fin };
}

function gsEsPalabra(plano, idx) {
  var c = plano.charCodeAt(idx);
  if ((c >= 97 && c <= 122) || (c >= 48 && c <= 57)) return true;
  if (c < 128) return false;
  return /[\p{L}\p{N}]/u.test(plano.charAt(idx));
}

/** Divide el texto plegado en tokens (rachas de letras o dígitos). Devuelve tokens y sus posiciones plegadas. */
function gsTokenizar(plano) {
  var toks = [], ta = [], tb = [];
  var n = plano.length, i = 0;
  while (i < n) {
    if (!gsEsPalabra(plano, i)) { i++; continue; }
    var j = i + 1;
    while (j < n && gsEsPalabra(plano, j)) j++;
    toks.push(plano.slice(i, j)); ta.push(i); tb.push(j);
    i = j;
  }
  return { toks: toks, ta: ta, tb: tb };
}

/** Prepara un texto para buscar en él: plegado + mapa de posiciones + tokens. Se calcula UNA vez al indexar. */
function gsPrepararSegmento(texto) {
  var f = gsPlegar(texto);
  var t = gsTokenizar(f.plano);
  return { texto: String(texto == null ? '' : texto), plano: f.plano, ini: f.ini, fin: f.fin, toks: t.toks, ta: t.ta, tb: t.tb };
}

function gsFallo(codigo, error) { return { ok: false, codigo: codigo, error: error }; }

function gsEsNumero(t) { return /^[0-9]+$/.test(t); }

/** Si la consulta completa es un DUI (9 dígitos, 6-8 con dígito verificador separado, u 8-9 palabras-dígito) devuelve sus 9 dígitos. */
function gsDuiDesdeConsulta(texto) {
  var s = String(texto).trim().toLowerCase();
  if (/^[0-9\s\-–—]+$/.test(s)) {
    var d = s.replace(/\D/g, '');
    if (d.length === 9) return d;
    if (/^[0-9]{1,8}\s*[-–—\s]\s*[0-9]$/.test(s) && d.length >= 6 && d.length <= 8) return d.padStart(9, '0');
    return null;
  }
  var ws = s.split(/[\s\-–—]+/).filter(function (w) { return w.length > 0; });
  if (ws.length >= 8 && ws.length <= 9 && ws.every(function (w) { return GS_PALABRAS_DIGITO[w] !== undefined; })) {
    var dd = ws.map(function (w) { return GS_PALABRAS_DIGITO[w]; }).join('');
    return dd.length === 8 ? '0' + dd : dd;
  }
  return null;
}

function gsCriterioDui(d9) {
  var palabras = d9.split('').map(function (x) { return GS_DIGITO_PALABRA[Number(x)]; });
  return {
    tipo: 'alt', req: true, etiqueta: 'DUI',
    opciones: [[d9.slice(0, 8), d9.slice(8)], [d9], palabras]
  };
}

/**
 * Analiza la consulta del usuario. Reconoce: términos sueltos (todos deben aparecer), frases exactas
 * entre comillas, números de decreto en cualquier formato ("Decreto No. 690", "decreto 690", "D. 690",
 * "d690") y DUI. Aplica límites estrictos de longitud y de cantidad de términos.
 */
function gsAnalizarConsulta(entrada, lim, dui9) {
  lim = lim || GS_LIM;
  if (typeof entrada !== 'string') return gsFallo('tipo', 'La consulta debe ser un texto.');
  if (entrada.length > lim.maxChars * 10) return gsFallo('larga', 'La consulta admite ' + lim.maxChars + ' caracteres como máximo. Acórtala y vuelve a intentar.');
  var texto = entrada.replace(/[\u0000-\u001f\u007f]/g, ' ').replace(/\s+/g, ' ').trim();
  if (!texto) return gsFallo('vacia', 'Escribe un DUI, un nombre, un número de decreto o una frase para buscar en el archivo.');
  if (texto.length > lim.maxChars) return gsFallo('larga', 'La consulta admite ' + lim.maxChars + ' caracteres como máximo. Acórtala y vuelve a intentar.');
  if (texto.length < 3) return gsFallo('corta', 'Escribe al menos 3 caracteres para buscar.');

  var d9 = dui9 || gsDuiDesdeConsulta(texto);
  if (d9) {
    return { ok: true, texto: texto, criterios: [gsCriterioDui(d9)], descripcion: [{ tipo: 'dui', texto: d9.slice(0, 8) + '-' + d9.slice(8) }], dui: d9 };
  }

  // 1) Separar grupos entre comillas y sin comillas
  var grupos = [], cur = '', dentro = false;
  for (var i = 0; i < texto.length; i++) {
    var ch = texto.charAt(i);
    if (GS_COMILLAS.indexOf(ch) >= 0) {
      if (cur.trim()) grupos.push({ q: dentro, raw: cur.trim() });
      cur = ''; dentro = !dentro;
    } else cur += ch;
  }
  if (cur.trim()) grupos.push({ q: dentro, raw: cur.trim() });

  var totalTokens = 0;
  var criterios = [], descripcion = [];
  var palabras = []; // palabras sin comillas: { raw, toks }
  var fraseOrden = []; // todos los tokens sin comillas, en orden (para el bono de frase implícita)
  function tokensDe(raw) {
    var t = gsTokenizar(gsPlegar(raw).plano).toks;
    totalTokens += t.length;
    return t;
  }
  var errorToken = null;
  function revisar(t) {
    for (var x = 0; x < t.length; x++) if (t[x].length > lim.maxLongToken) errorToken = t[x];
  }
  for (var g = 0; g < grupos.length; g++) {
    var gr = grupos[g];
    if (gr.q) {
      var tq = tokensDe(gr.raw); revisar(tq);
      if (tq.length === 0) continue;
      if (tq.length === 1) criterios.push({ tipo: 'termino', t: tq[0], pref: false, req: true });
      else criterios.push({ tipo: 'frase', toks: tq, pref: false, req: true });
      descripcion.push({ tipo: tq.length === 1 ? 'termino' : 'frase', texto: gr.raw });
    } else {
      var ps = gr.raw.split(' ');
      for (var w = 0; w < ps.length; w++) {
        if (!ps[w]) continue;
        var tw = tokensDe(ps[w]); revisar(tw);
        if (tw.length > 0) palabras.push({ raw: ps[w], toks: tw });
      }
    }
    if (totalTokens > lim.maxTokens) break;
  }
  if (errorToken !== null) return gsFallo('token_largo', 'Una de las palabras es demasiado larga (máximo ' + lim.maxLongToken + ' letras). Revisa lo que escribiste.');
  if (totalTokens > lim.maxTokens) return gsFallo('muchos_terminos', 'Escribiste demasiadas palabras: el máximo es ' + lim.maxTokens + '. Usa menos palabras o agrúpalas entre comillas para buscar una frase exacta.');

  // 2) Números de decreto: clave (decreto, d, acuerdo, ley...) + relleno opcional (no, n°, legislativo) + número
  var plano = []; // { t, w }
  for (var a = 0; a < palabras.length; a++) for (var b = 0; b < palabras[a].toks.length; b++) plano.push({ t: palabras[a].toks[b], w: a, usado: false });
  for (var p = 0; p < plano.length; p++) {
    var tk = plano[p].t;
    if (plano[p].usado) continue;
    var pegado = /^(decreto|decr|dec|acuerdo|acdo|ley|d)([0-9]{1,5})$/.exec(tk);
    if (pegado) {
      plano[p].usado = true;
      criterios.push({ tipo: 'decreto', numero: pegado[2], req: true });
      descripcion.push({ tipo: 'decreto', texto: 'Decreto N.º ' + pegado[2] });
      continue;
    }
    if (GS_CLAVES_DOC[tk] === 1) {
      var q = p + 1;
      while (q < plano.length && GS_RELLENO_DOC[plano[q].t] === 1) q++;
      if (q < plano.length && /^[0-9]{1,5}$/.test(plano[q].t)) {
        for (var z = p; z <= q; z++) plano[z].usado = true;
        criterios.push({ tipo: 'decreto', numero: plano[q].t, req: true });
        descripcion.push({ tipo: 'decreto', texto: 'Decreto N.º ' + plano[q].t });
      }
    } else if (GS_RELLENO_DOC[tk] === 1) {
      // "No. 690" sin clave: se descarta el relleno y queda el número como término
      var r = p + 1;
      while (r < plano.length && GS_RELLENO_DOC[plano[r].t] === 1) r++;
      if (r < plano.length && /^[0-9]{1,5}$/.test(plano[r].t)) for (var y = p; y < r; y++) plano[y].usado = true;
    }
  }
  // 3) Palabras restantes: un token = término; varios tokens (p. ej. "s.a.") = frase
  var restantes = [];
  for (var u = 0; u < palabras.length; u++) restantes.push([]);
  for (var v = 0; v < plano.length; v++) if (!plano[v].usado) restantes[plano[v].w].push(plano[v].t);
  var terminosSueltos = [];
  for (var k = 0; k < palabras.length; k++) {
    var rt = restantes[k];
    if (rt.length === 0) continue;
    if (rt.length === 1) {
      var t1 = rt[0];
      fraseOrden.push(t1);
      if (!gsEsNumero(t1) && t1.length < 2) continue;
      terminosSueltos.push({ t: t1, raw: palabras[k].raw });
    } else {
      for (var m = 0; m < rt.length; m++) fraseOrden.push(rt[m]);
      criterios.push({ tipo: 'frase', toks: rt, pref: !gsEsNumero(rt[rt.length - 1]) && rt[rt.length - 1].length >= 3, req: true });
      descripcion.push({ tipo: 'frase', texto: palabras[k].raw });
    }
  }
  var utiles = terminosSueltos.filter(function (x) { return GS_VACIAS[x.t] !== 1; });
  for (var s = 0; s < utiles.length; s++) {
    var tt = utiles[s].t;
    criterios.push({ tipo: 'termino', t: tt, pref: !gsEsNumero(tt) && tt.length >= 3, req: true });
    descripcion.push({ tipo: 'termino', texto: utiles[s].raw });
  }
  var requeridos = criterios.filter(function (c) { return c.req; }).length;
  if (requeridos === 0) return gsFallo('sin_terminos', 'Tu consulta solo tiene palabras muy cortas o comunes (como "de" o "la"). Agrega un nombre, un número o una palabra más específica.');
  if (requeridos > lim.maxCriterios) return gsFallo('muchos_terminos', 'Escribiste demasiados términos distintos: el máximo es ' + lim.maxCriterios + '. Quita alguno o agrúpalos entre comillas.');

  // 4) Bono de frase implícita: los mismos términos juntos y en el mismo orden puntúan más (no se exigen)
  if (fraseOrden.length >= 2 && criterios.every(function (c) { return c.tipo !== 'decreto'; })) {
    criterios.push({ tipo: 'frase', toks: fraseOrden, pref: false, req: false });
  }
  return { ok: true, texto: texto, criterios: criterios, descripcion: descripcion, dui: null };
}

/**
 * Coincidencia de palabra con tolerancia de flexión: la palabra puede llevar hasta 2 letras más que el término
 * (plural, género: "tributaria" -> "tributarias"), pero NO es una búsqueda por prefijo libre: "tránsito" no coincide
 * con "transitoria". Así lo remarcado es siempre la palabra completa que el usuario reconoce como coincidencia.
 */
function gsPrefijo(tok, t) { return tok.length >= t.length && tok.length <= t.length + 2 && tok.lastIndexOf(t, 0) === 0; }

function gsSoloSeparador(plano, a, b) {
  if (b - a > 4) return false;
  for (var i = a; i < b; i++) {
    if (' .-:°º,'.indexOf(plano.charAt(i)) < 0) return false;
  }
  return true;
}

/**
 * Una frase exacta solo puede cruzar separadores "blandos" (espacio, coma, guion, punto de abreviatura): nunca un fin de
 * oración ("ley. Otra"), dos puntos, punto y coma, paréntesis ni comillas. Así "ley otra" no coincide con "la ley. Otra cosa".
 */
function gsGapFrase(plano, a, b, previo) {
  if (b - a > 5) return false;
  var punto = false;
  for (var i = a; i < b; i++) {
    var ch = plano.charAt(i);
    if (" .,-'/°".indexOf(ch) < 0) return false;
    if (ch === '.') punto = true;
  }
  // Punto seguido de espacio = fin de oración, salvo tras abreviatura, letra suelta o número ("S.A. de", "No. 690")
  if (punto && plano.slice(a, b).indexOf('. ') >= 0 && previo.length > 1 && GS_ABREV[previo] !== 1 && !/^[0-9]+$/.test(previo)) return false;
  return true;
}

function gsHit(seg, c, a, b, fuerte) {
  return { c: c, a: seg.ini[a], b: seg.fin[b - 1], fuerte: fuerte };
}

function gsSecuencia(seg, i, seq, pref) {
  var k = seq.length;
  if (i + k > seg.toks.length) return false;
  for (var j = 0; j < k; j++) {
    var tk = seg.toks[i + j];
    if (j === k - 1 && pref) { if (!gsPrefijo(tk, seq[j])) return false; }
    else if (tk !== seq[j]) return false;
    if (j > 0 && !gsGapFrase(seg.plano, seg.tb[i + j - 1], seg.ta[i + j], seg.toks[i + j - 1])) return false;
  }
  return true;
}

/**
 * Localiza todos los criterios dentro de un segmento preparado. Devuelve coincidencias
 * { c: índice de criterio, a, b: rango en el texto ORIGINAL, fuerte: frase/número de decreto }.
 */
function gsLocalizar(seg, criterios) {
  var hits = [];
  var toks = seg.toks, n = toks.length;
  for (var c = 0; c < criterios.length; c++) {
    var cr = criterios[c];
    if (cr.tipo === 'termino') {
      for (var i = 0; i < n; i++) {
        if (cr.pref ? gsPrefijo(toks[i], cr.t) : toks[i] === cr.t) {
          hits.push(gsHit(seg, c, seg.ta[i], seg.tb[i], false));
        }
      }
    } else if (cr.tipo === 'frase') {
      var k = cr.toks.length;
      for (var i2 = 0; i2 + k <= n; i2++) {
        if (gsSecuencia(seg, i2, cr.toks, cr.pref)) {
          var ult = i2 + k - 1;
          hits.push(gsHit(seg, c, seg.ta[i2], seg.tb[ult], true));
        }
      }
    } else if (cr.tipo === 'alt') {
      for (var o = 0; o < cr.opciones.length; o++) {
        var op = cr.opciones[o], ko = op.length;
        for (var i3 = 0; i3 + ko <= n; i3++) {
          if (gsSecuencia(seg, i3, op, false)) hits.push(gsHit(seg, c, seg.ta[i3], seg.tb[i3 + ko - 1], true));
        }
      }
    } else if (cr.tipo === 'decreto') {
      for (var i4 = 0; i4 < n; i4++) {
        var tk = toks[i4];
        if (tk === cr.numero) {
          var j = i4 - 1;
          while (j >= 0 && GS_RELLENO_DOC[toks[j]] === 1 && gsSoloSeparador(seg.plano, seg.tb[j], seg.ta[j + 1])) j--;
          if (j >= 0 && GS_CLAVES_DOC[toks[j]] === 1 && gsSoloSeparador(seg.plano, seg.tb[j], seg.ta[j + 1])) {
            hits.push(gsHit(seg, c, seg.ta[j], seg.tb[i4], true));
          }
        } else {
          var m = /^(decreto|decr|dec|acuerdo|acdo|ley|d)([0-9]{1,5})$/.exec(tk);
          if (m && m[2] === cr.numero) hits.push(gsHit(seg, c, seg.ta[i4], seg.tb[i4], true));
        }
      }
    }
  }
  return hits;
}

/** Une SOLO rangos que se solapan o se tocan: dos apariciones separadas siguen siendo dos marcas distintas. */
function gsUnirRangos(rangos, texto) {
  var r = rangos.map(function (x) { return [x[0], x[1]]; }).sort(function (x, y) { return x[0] - y[0] || x[1] - y[1]; });
  var out = [];
  for (var i = 0; i < r.length; i++) {
    var ult = out.length ? out[out.length - 1] : null;
    if (ult && r[i][0] <= ult[1]) {
      if (r[i][1] > ult[1]) ult[1] = r[i][1];
    } else out.push(r[i]);
  }
  return out;
}

/** Texto -> HTML seguro: escapa SIEMPRE el texto y solo entonces inserta <mark>. */
function gsResaltarHtml(texto, rangos, clase, atributos) {
  texto = String(texto == null ? '' : texto);
  var abre = '<mark class="' + gsEscapar(clase || 'gs-mark') + '"' + (atributos || '') + '>';
  var html = '', pos = 0;
  for (var i = 0; i < rangos.length; i++) {
    var a = Math.max(pos, rangos[i][0]), b = Math.min(texto.length, rangos[i][1]);
    if (b <= a) continue;
    html += gsEscapar(texto.slice(pos, a)) + abre + gsEscapar(texto.slice(a, b)) + '</mark>';
    pos = b;
  }
  return html + gsEscapar(texto.slice(pos));
}

/** Atajo para el navegador: resalta una consulta completa dentro de un texto. */
function gsResaltarConsulta(texto, consulta, clase, atributos) {
  var an = gsAnalizarConsulta(consulta);
  if (!an.ok) return gsEscapar(texto);
  var seg = gsPrepararSegmento(texto);
  var rangos = gsLocalizar(seg, an.criterios).map(function (h) { return [h.a, h.b]; });
  return gsResaltarHtml(texto, gsUnirRangos(rangos, texto), clase, atributos);
}

var GL_GAP_RUN = 14;      // separación (pt) que parte una línea en tramos independientes
var GL_MARGEN_CAB = 62;   // alto (pt) de la franja de cabecera y pie

function glModeloPagina(items, anchoPag, altoPag) {
  var mid = anchoPag / 2;
  var elems = [];
  for (var i = 0; i < items.length; i++) {
    var it = items[i];
    if (!it || typeof it.str !== 'string' || it.str === '') continue;
    var t = it.transform;
    // Texto girado (marca de agua diagonal): no forma parte del contenido.
    if (Math.abs(t[1]) > 0.01 || Math.abs(t[2]) > 0.01) continue;
    elems.push({ i: i, x: t[4], y: t[5], w: it.width || 0, h: Math.abs(t[3]) || 8, s: it.str });
  }
  // Agrupar por línea base (tolerancia pequeña) de arriba hacia abajo.
  elems.sort(function (p, q) { return q.y - p.y || p.x - q.x; });
  var filas = [];
  for (var k = 0; k < elems.length; k++) {
    var e = elems[k];
    var f = filas.length ? filas[filas.length - 1] : null;
    // Un superíndice (6ª, 2°) queda más arriba y con letra menor: cuenta como parte de la misma línea.
    var tol = Math.max(1.2, 0.4 * Math.min(f ? f.h : 8, e.h));
    if (f && Math.min(f.h, e.h) < 0.8 * Math.max(f.h, e.h)) tol = Math.min(4.5, 0.6 * Math.max(f.h, e.h));
    if (f && Math.abs(f.y - e.y) <= tol) { f.e.push(e); if (e.h > f.h) f.h = e.h; }
    else filas.push({ y: e.y, h: e.h, e: [e] });
  }
  // Cada fila se parte en tramos por separación horizontal.
  var tramos = [];
  for (var r = 0; r < filas.length; r++) {
    var fila = filas[r];
    fila.e.sort(function (p, q) { return p.x - q.x; });
    var cur = null;
    for (var j = 0; j < fila.e.length; j++) {
      var el = fila.e[j];
      var vacio = el.s.trim() === '';
      if (cur && el.x - cur.x1 > GL_GAP_RUN && !vacio) { tramos.push(cur); cur = null; }
      if (!cur) { if (vacio) continue; cur = { y: fila.y, h: fila.h, x0: el.x, x1: el.x + el.w, e: [] }; }
      cur.e.push(el);
      cur.x1 = Math.max(cur.x1, el.x + el.w);
    }
    if (cur) tramos.push(cur);
  }
  // Clasificar cada tramo.
  for (var m = 0; m < tramos.length; m++) {
    var tr = tramos[m];
    var ancho = tr.x1 - tr.x0;
    var cruza = tr.x0 < mid - 18 && tr.x1 > mid + 18;
    tr.col = cruza ? 0 : ((tr.x0 + tr.x1) / 2 < mid ? 1 : 2);
    var txtTramo = '';
    for (var w = 0; w < tr.e.length; w++) txtTramo += tr.e[w].s;
    // Cabecera de la página, o el pie con la dirección de la Imprenta (solo en la portada).
    var cab = tr.y > altoPag - GL_MARGEN_CAB || (tr.y < 62 && /^\s*Direcci[oó]n:/.test(txtTramo));
    // Título o franja: cruza el centro de la página y está centrado y corto. Un párrafo a todo el ancho es cuerpo.
    var centrado = cruza && Math.abs((tr.x0 + tr.x1) / 2 - mid) < 30 && ancho < 330;
    tr.k = cab ? 'c' : (centrado ? 'b' : 'p');
    tr.ancho = ancho;
  }
  // Orden de lectura: cabecera, luego bandas delimitadas por los tramos a todo el ancho.
  tramos.sort(function (p, q) { return q.y - p.y || p.x0 - q.x0; });
  var orden = [];
  var cab_ = [], pie_ = [];
  var banda = { L: [], R: [] };
  function volcar() {
    for (var a = 0; a < banda.L.length; a++) orden.push(banda.L[a]);
    for (var b = 0; b < banda.R.length; b++) orden.push(banda.R[b]);
    banda = { L: [], R: [] };
  }
  for (var n = 0; n < tramos.length; n++) {
    var t2 = tramos[n];
    if (t2.k === 'c') { (t2.y > altoPag / 2 ? cab_ : pie_).push(t2); continue; }
    if (t2.col === 0) { volcar(); orden.push(t2); }
    else if (t2.col === 1) banda.L.push(t2);
    else banda.R.push(t2);
  }
  volcar();
  var todos = cab_.concat(orden, pie_);

  var texto = '';
  var ci = [], co = [], lineas = [];
  for (var q = 0; q < todos.length; q++) {
    var T = todos[q];
    var a0 = texto.length;
    var prev = null;
    for (var z = 0; z < T.e.length; z++) {
      var it2 = T.e[z];
      if (prev && it2.x - (prev.x + prev.w) > 0.2 * T.h && !/\s$/.test(prev.s) && !/^\s/.test(it2.s)) {
        texto += ' '; ci.push(-1); co.push(0);
      }
      for (var c = 0; c < it2.s.length; c++) { ci.push(it2.i); co.push(c); }
      texto += it2.s;
      prev = it2;
    }
    lineas.push({ a: a0, b: texto.length, k: T.k, c: T.col, x0: Math.round(T.x0 * 10) / 10, x1: Math.round(T.x1 * 10) / 10, y: Math.round(T.y * 10) / 10 });
    if (q < todos.length - 1) { texto += '\n'; ci.push(-1); co.push(0); }
  }
  return { texto: texto, ci: ci, co: co, lineas: lineas };
}

/**
 * Texto para buscar: une las palabras partidas con guion al final de línea ("ESCO-\nBAR" -> "ESCOBAR").
 * Devuelve el texto y, por cada carácter, su posición en el texto original de la página.
 */
function glTextoBusqueda(texto) {
  var out = [], mapa = [];
  var n = texto.length;
  for (var i = 0; i < n; i++) {
    var ch = texto.charAt(i);
    if (ch === '-' && texto.charAt(i + 1) === '\n' && i > 0 && /\p{L}/u.test(texto.charAt(i - 1)) && /\p{L}/u.test(texto.charAt(i + 2) || '')) { i++; continue; }
    out.push(ch); mapa.push(i);
  }
  mapa.push(n);
  return { t: out.join(''), mapa: mapa };
}

      if (token !== VZ.carga || P.estado !== 1) return;
      var divs = [], strs = [];
      P.tareaTexto = window.pdfjsLib.renderTextLayer({ textContentSource: tc, container: capaTxt, viewport: vp, textDivs: divs, textContentItemsStr: strs });
      return P.tareaTexto.promise.then(function () {
        P.divs = divs;
        P.modelo = glModeloPagina(tc.items, vp1.width, vp1.height);
        // El texto de la página debe ser el mismo que el del índice; si difiere se reindexa solo esa página.
        if (VZ.textos[n - 1] !== P.modelo.texto) { VZ.textos[n - 1] = P.modelo.texto; VZ.busq[n] = null; vzRecalcular(true); }
      });
    }).catch(function (e) { if (!e || e.name !== 'RenderingCancelledException') console.warn('Texto de la página ' + n, e); });
    return Promise.all([pRender, pTexto]).then(function () {
      if (token !== VZ.carga || P.estado !== 1) return;
      P.estado = 2;
      vzPintar(n);
    });
  });
}

// ---------------------------------------------------------------- geometría exacta de un rango de texto
function vzRectsRango(P, a, b) {
  var m = P.modelo, out = [];
  if (!m || !P.divs) return out;
  var base = P.el.getBoundingClientRect();
  var i = a;
  while (i < b) {
    var ci = m.ci[i];
    if (ci < 0) { i++; continue; }
    var j = i;
    while (j < b && m.ci[j] === ci) j++;
    var div = P.divs[ci];
    var nodo = div && div.firstChild;
    if (nodo && nodo.nodeType === 3) {
      var r = document.createRange();
      var ini = Math.min(m.co[i], nodo.length), fin = Math.min(nodo.length, m.co[j - 1] + 1);
      if (fin > ini) {
        r.setStart(nodo, ini); r.setEnd(nodo, fin);
        var rs = r.getClientRects();
        for (var k = 0; k < rs.length; k++) {
          var q = rs[k];
          if (q.width < 0.5 || q.height < 0.5) continue;
          out.push({ l: q.left - base.left, t: q.top - base.top, w: q.width, h: q.height });
        }
      }
    }
    i = j;
  }
  return out;
}

function vzFundirRects(rs) {
  rs.sort(function (p, q) { return Math.abs(p.t - q.t) < 2 ? p.l - q.l : p.t - q.t; });
  var out = [];
  rs.forEach(function (r) {
    var u = out[out.length - 1];
    if (u && Math.abs(u.t - r.t) < 2.5 && Math.abs(u.h - r.h) < 3 && r.l - (u.l + u.w) < 4) {
      var fin = Math.max(u.l + u.w, r.l + r.w); u.l = Math.min(u.l, r.l); u.w = fin - u.l;
    } else out.push({ l: r.l, t: r.t, w: r.w, h: r.h });
  });
  return out;
}

function vzDibujar(capa, rects, clase) {
  rects.forEach(function (r) {
    var d = document.createElement('div');
    d.className = clase;
    d.style.cssText = 'left:' + r.l.toFixed(1) + 'px;top:' + r.t.toFixed(1) + 'px;width:' + r.w.toFixed(1) + 'px;height:' + r.h.toFixed(1) + 'px';
    capa.appendChild(d);
  });
}

function vzPintar(n) {
  var P = VZ.paginas[n];
  if (!P || P.estado !== 2 || !P.capaHl || !P.modelo) return;
  P.capaHl.innerHTML = '';
  // 1) publicación completa seleccionada (puede abarcar varias páginas)
  if (VZ.regActivo && VZ.regActivo.rangos) {
    var rs = [];
    VZ.regActivo.rangos.forEach(function (g) { if (g[0] === n) rs = rs.concat(vzRectsRango(P, g[1], g[2])); });
    var fund = vzFundirRects(rs);
    vzDibujar(P.capaHl, fund, 'vz-reg');
    if (fund.length) {
      var top = Math.min.apply(null, fund.map(function (r) { return r.t; })), bot = Math.max.apply(null, fund.map(function (r) { return r.t + r.h; }));
      var izq = Math.min.apply(null, fund.map(function (r) { return r.l; }));
      vzDibujar(P.capaHl, [{ l: izq - 5, t: top, w: 0, h: bot - top }], 'vz-reg-borde');
    }
  }
  // 2) coincidencias de la búsqueda
  var ms = VZ.matches;
  for (var i = 0; i < ms.length; i++) {
    if (ms[i].p !== n) continue;
    var fr = vzFundirRects(vzRectsRango(P, ms[i].a, ms[i].b));
    vzDibujar(P.capaHl, fr, 'vz-hl' + (i === VZ.idx ? ' vz-act' : ''));
    ms[i].pintado = true;
  }
}

function vzRepintarTodo() { for (var n = 1; n < VZ.paginas.length; n++) vzPintar(n); }

// ---------------------------------------------------------------- búsqueda (como la del navegador)
function vzPrep(n) {
  if (VZ.busq[n]) return VZ.busq[n];
  var txt = VZ.textos[n - 1] || '';
  var tb = glTextoBusqueda(txt);
  VZ.busq[n] = { t: tb.t, mapa: tb.mapa, prep: gsPrepararSegmento(tb.t) };
  return VZ.busq[n];
}

function vzAPagina(B, a, b) {
  // [a,b) sobre el texto de búsqueda -> [A,B) sobre el texto de la página
  return [B.mapa[a], B.mapa[b - 1] + 1];
}

function vzBuscarPagina(n, fq, an) {
  var B = vzPrep(n), out = [];
  if (!B.t) return out;
  if (fq) {
    // fq es una expresión de palabras separadas por cualquier signo: "jose alejandro rivas" encuentra "JOSÉ ALEJANDRO RIVAS."
    var re = new RegExp(fq, 'g'), x;
    var plano = B.prep.plano;
    while ((x = re.exec(plano)) !== null) {
      if (x[0].length === 0) { re.lastIndex++; continue; }
      var a = B.prep.ini[x.index], b = B.prep.fin[x.index + x[0].length - 1];
      var r = vzAPagina(B, a, b);
      out.push({ p: n, a: r[0], b: r[1] });
    }
  } else if (an) {
    var rangos = gsLocalizar(B.prep, an.criterios).map(function (h) { return [h.a, h.b]; });
    gsUnirRangos(rangos, B.t).forEach(function (g) {
      var r2 = vzAPagina(B, g[0], g[1]);
      out.push({ p: n, a: r2[0], b: r2[1] });
    });
  }
  return out;
}

function vzRecalcular(mantener) {
  var term = (VZ.term || '').replace(/\s+/g, ' ').trim();
  var previo = VZ.idx >= 0 && VZ.matches[VZ.idx] ? VZ.matches[VZ.idx] : null;
  VZ.matches = [];
  if (term.length >= 2 && VZ.diario) {
    var d9 = (typeof gsDuiDesdeConsulta === 'function') ? gsDuiDesdeConsulta(term) : null;
    var esDui = !!d9 || /^\d[\d\s-]{6,}$/.test(term);
    var palabras = esDui ? [] : gsPlegar(term).plano.split(/[^a-z0-9]+/).filter(Boolean);
    var fq = palabras.length ? palabras.join('[^a-z0-9]+') : '';
    var an = null;
    if (esDui) { an = gsAnalizarConsulta(term, undefined, d9 || undefined); if (!an.ok) an = null; }
    var total = VZ.diario.paginas, n;
    if (fq) {
      for (n = 1; n <= total; n++) VZ.matches = VZ.matches.concat(vzBuscarPagina(n, fq, null));
      // Sin coincidencia literal de la frase: se marcan sus palabras por separado (≥ 4 letras), nunca el diario entero.
      if (VZ.matches.length === 0) {
        var an2 = gsAnalizarConsulta(term);
        if (an2.ok) for (n = 1; n <= total; n++) VZ.matches = VZ.matches.concat(vzBuscarPagina(n, '', an2));
      }
    } else if (an) {
      for (n = 1; n <= total; n++) VZ.matches = VZ.matches.concat(vzBuscarPagina(n, '', an));
    }
  }
  VZ.idx = -1;
  if (mantener && previo) {
    for (var i = 0; i < VZ.matches.length; i++) if (VZ.matches[i].p === previo.p && VZ.matches[i].a === previo.a) { VZ.idx = i; break; }
  }
  vzContador();
  vzRepintarTodo();
}

function vzContador() {
  var el = vzEl('pdf-highlight-counter');
  el.textContent = VZ.matches.length ? ((VZ.idx >= 0 ? VZ.idx + 1 : 0) + '/' + VZ.matches.length) : '0/0';
  var campo = vzEl('ebook-inline-search');
  campo.style.color = (VZ.term && !VZ.matches.length) ? '#fca5a5' : '';
}

function vzIrCoincidencia(i, opts) {
  if (!VZ.matches.length) { vzAviso(VZ.term ? 'No hay coincidencias en este diario.' : 'Escribe qué buscar en el diario.'); return Promise.resolve(); }
  i = ((i % VZ.matches.length) + VZ.matches.length) % VZ.matches.length;
  var viejo = VZ.idx >= 0 ? VZ.matches[VZ.idx] : null;
  VZ.idx = i;
  var m = VZ.matches[i];
  vzContador();
  if (viejo && viejo.p !== m.p) vzPintar(viejo.p);
  return vzScrollAPagina(m.p, 0).then(function () { return vzRender(m.p); }).then(function () {
    vzPintar(m.p);
    var P = VZ.paginas[m.p];
    var act = P && P.capaHl && P.capaHl.querySelector('.vz-act');
    if (act) {
      var cont = vzEl('vz-scroll');
      var y = P.el.offsetTop + act.offsetTop - cont.clientHeight * 0.35;
      cont.scrollTo({ top: Math.max(0, y), behavior: (opts && opts.suave) ? 'smooth' : 'auto' });
    }
  });
}

// ---------------------------------------------------------------- navegación
function vzScrollAPagina(n, margen) {
  var P = VZ.paginas[n];
  if (!P) return Promise.resolve();
  vzEl('vz-scroll').scrollTo({ top: Math.max(0, P.el.offsetTop - (margen == null ? 10 : margen)), behavior: 'auto' });
  vzActualizarPaginaActual();
  return Promise.resolve();
}

function vzIrPagina(n) {
  n = Math.max(1, Math.min(VZ.diario ? VZ.diario.paginas : 1, Math.floor(Number(n) || 1)));
  return vzScrollAPagina(n).then(function () { return vzRender(n); });
}

function vzPaginaVisible() {
  var cont = vzEl('vz-scroll'), P1 = VZ.paginas[1];
  if (!P1) return 1;
  var paso = P1.el.offsetHeight + VZ_GAP;
  var y = cont.scrollTop + cont.clientHeight * 0.3 - P1.el.offsetTop;
  return Math.max(1, Math.min(VZ.paginas.length - 1, Math.floor(y / paso) + 1));
}

function vzActualizarPaginaActual() {
  var n = vzPaginaVisible();
  if (n === VZ.paginaActual && vzEl('ebook-page-number').value === String(n)) return;
  VZ.paginaActual = n;
  vzEl('ebook-page-number').value = n;
  vzEl('ebook-top-page-badge').textContent = 'Pág. ' + n + ' de ' + VZ.diario.paginas;
  vzEl('sidebar-page-label').textContent = '(Pág. ' + n + ')';
  vzPanel();
}

function vzPanel() {
  var lista = vzEl('vz-panel-lista');
  var n = VZ.paginaActual;
  var regs = VZ.registros.filter(function (r) { return r.pagina <= n && (r.paginaFin || r.pagina) >= n; });

      (VZ.sinTexto[n] ? 'Esta página no tiene texto digital en el PDF, por eso no hay transcripción. Se muestra solo el documento original.' : 'No hay publicaciones indexadas en esta página. Usa el buscador del visor para localizar cualquier palabra.') + '</p></div>';
    return;
  }
  lista.innerHTML = regs.map(function (r) {

    var cuerpo = VZ.term && VZ.term.length >= 2 ? gsResaltarConsulta(r.contenido, VZ.term, 'vz-m', '') : gsEscapar(r.contenido);

    return '<article class="vz-tarjeta' + (activa ? ' vz-activa' : '') + '" data-vz-reg="' + gsEscapar(r.id) + '">' +
      '<div class="flex flex-wrap items-center justify-between gap-2 mb-2"><span class="text-[11px] font-mono font-bold text-gs-accent">' + gsEscapar(vzNombreClasif(r)) + '</span>' +
      '<span class="text-[11px] font-mono font-bold text-gs-warning">' + gsEscapar(rango) + '</span></div>' +
      '<h6 class="font-bold text-gs-text text-[12px] leading-snug mb-2">' + gsEscapar(r.titular) + '</h6>' +
      '<div class="text-[11.5px] text-gs-text2 whitespace-pre-line break-words max-h-64 overflow-y-auto pr-1">' + cuerpo + '</div>' +
      (duis ? '<div class="flex flex-wrap items-center gap-1.5 mt-2.5 pt-2 border-t border-gs-border"><span class="text-[10px] text-gs-text3 font-mono">DUI:</span>' + duis + '</div>' : '') +
      '<div class="mt-2.5 flex gap-2"><button type="button" class="vz-btn' + (activa ? ' vz-btn-primario' : '') + '" data-vz-resaltar="' + gsEscapar(r.id) + '"><i class="fa-solid fa-highlighter"></i> ' + (activa ? 'Quitar resaltado' : 'Resaltar en el PDF') + '</button></div></article>';
  }).join('');
  var act = lista.querySelector('.vz-activa');
  if (act) lista.scrollTop = act.offsetTop - 8;
}

function vzResaltarRegistro(id, sinScroll) {
  var r = VZ.registros.filter(function (x) { return x.id === id; })[0];
  var igual = VZ.regActivo && VZ.regActivo.id === id;
  VZ.regActivo = igual && !sinScroll ? null : (r || null);
  vzRepintarTodo();
  vzPanel();
  if (VZ.regActivo && !sinScroll) {
    var P0 = VZ.regActivo.rangos && VZ.regActivo.rangos[0] ? VZ.regActivo.rangos[0][0] : VZ.regActivo.pagina;
    var ult = VZ.regActivo.rangos && VZ.regActivo.rangos.length ? VZ.regActivo.rangos[VZ.regActivo.rangos.length - 1][0] : P0;
    return vzScrollAPagina(P0, 10).then(function () {
      var ps = [];
      for (var n = P0; n <= ult; n++) ps.push(vzRender(n));
      return Promise.all(ps);
    }).then(function () {
      vzRepintarTodo();
      var Pg = VZ.paginas[P0], b = Pg && Pg.capaHl && Pg.capaHl.querySelector('.vz-reg-borde');
      if (b) vzEl('vz-scroll').scrollTo({ top: Math.max(0, Pg.el.offsetTop + b.offsetTop - 90), behavior: 'auto' });
      if (ult !== P0) vzAviso('Esta publicación continúa hasta la pág. ' + ult + ': se resalta completa.');
    });
  }
  return Promise.resolve();
}

// ---------------------------------------------------------------- otros diarios y retorno
function vzOtros() {
  var cont = vzEl('vz-otros');
  var term = (VZ.term || '').trim();
  var back = VZ.pila.length ? VZ.pila[VZ.pila.length - 1] : null;
  var html = '';
  if (back) html += '<button type="button" class="vz-btn" data-vz-volver="1"><i class="fa-solid fa-arrow-left"></i> Volver al ' + gsEscapar(back.etiqueta) + ' (pág. ' + back.pagina + ')</button>';
  var finalizar = function (lista) {
    if (lista && lista.length) {
      html += '<span>Esta búsqueda también aparece en:</span>' + lista.map(function (o) {
        return '<button type="button" class="vz-chip" data-vz-diario="' + gsEscapar(o.id) + '" data-vz-pag="' + o.primeraPagina + '"><i class="fa-solid fa-book-open"></i>' + gsEscapar('Tomo ' + o.tomo + ' · N.° ' + o.numero + ' · ' + o.fecha) + ' <b>' + o.total + '</b></button>';
      }).join(' ');
    }
    cont.innerHTML = html;
    cont.classList.toggle('hidden', !html);
    cont.style.display = html ? 'flex' : 'none';
  };
  if (term.length < 3) { finalizar(null); return; }
  var clave = term.toLowerCase();
  var usar = function (data) {
    if (term !== (VZ.term || '').trim()) return;
    finalizar((data || []).filter(function (o) { return VZ.diario && o.id !== VZ.diario.id && o.total > 0; }));
  };
  if (VZ.cacheOtros[clave]) { usar(VZ.cacheOtros[clave]); return; }
  finalizar(null);
  fetch('/api/v1/coincidencias?q=' + encodeURIComponent(term)).then(function (r) { return r.json(); }).then(function (d) {
    VZ.cacheOtros[clave] = (d && d.diarios) || [];
    usar(VZ.cacheOtros[clave]);
  }).catch(function () {});
}

}

// ---------------------------------------------------------------- apertura
function vzAbrir(o) {
  o = o || {};
  var modal = vzEl('modal-ebook');
  modal.classList.remove('hidden'); modal.classList.add('flex');
  document.documentElement.style.overflow = 'hidden';
  var term = o.termino != null ? String(o.termino) : (VZ.term || '');
  return vzCatalogo().then(function (cat) {
    if (!cat.length) { vzEl('vz-estado').style.display = 'block'; vzEl('vz-estado').textContent = 'No hay diarios disponibles.'; return; }
    var id = o.diarioId && cat.some(function (c) { return c.id === o.diarioId; }) ? o.diarioId : (VZ.diario && !o.diarioId ? VZ.diario.id : cat[cat.length - 1].id);
    var carga = (!VZ.diario || VZ.diario.id !== id || !VZ.paginas.length) ? vzCargarDiario(id) : Promise.resolve();
    return carga.then(function () {
      VZ.term = term;
      vzEl('ebook-inline-search').value = term;
      VZ.regActivo = null;
      vzRecalcular();
      vzOtros();
      var pag = Math.max(1, Math.min(VZ.diario.paginas, Math.floor(Number(o.pagina) || 1)));
      var rec = o.registroId ? VZ.registros.filter(function (r) { return r.id === o.registroId; })[0] : null;
      var ini = rec ? rec.pagina : pag, fin = rec ? (rec.paginaFin || rec.pagina) : pag;
      var objetivo = -1;
      for (var i = 0; i < VZ.matches.length; i++) { if (VZ.matches[i].p >= ini && VZ.matches[i].p <= Math.max(fin, ini)) { objetivo = i; break; } }
      if (objetivo < 0) for (var k = 0; k < VZ.matches.length; k++) { if (VZ.matches[k].p >= pag) { objetivo = k; break; } }
      var listo = vzIrPagina(pag);
      if (rec && !VZ.matches.length) listo = listo.then(function () { return vzResaltarRegistro(rec.id, false); });
      else if (objetivo >= 0) listo = listo.then(function () { return vzIrCoincidencia(objetivo); });
      return listo.then(function () { vzActualizarPaginaActual(); vzPanel(); });
    });
  }).catch(function (e) {
    console.warn('Visor:', e);
    vzEl('vz-estado').style.display = 'block';
    vzEl('vz-estado').innerHTML = '<p class="text-gs-danger font-bold">No se pudo cargar el diario.</p><p class="mt-1">Revisa tu conexión y vuelve a intentar.</p>';
  });
}

    function gsMostrarFallo(contenedor, titulo, detalle, conReintento) {

    // Dirección compartible: ?q=...&categoria=...&forceDui=1&pagina=n
    function gsActualizarUrl(q, cat, forceDui, pag, reemplazar) {
      try {
        const p = new URLSearchParams();
        if (q) p.set('q', q);
        if (cat) p.set('categoria', cat);
        if (forceDui) p.set('forceDui', '1');
        if (pag > 1) p.set('pagina', String(pag));
        const destino = location.pathname + (p.toString() ? '?' + p.toString() : '');
        if (destino === location.pathname + location.search) return;
        if (reemplazar) history.replaceState(null, '', destino);
        else history.pushState(null, '', destino);
      } catch (_) { /* historial no disponible: la búsqueda funciona igual */ }
    }

    function searchDiario(e, pagina, sinHistorial) {
      if (e && e.preventDefault) e.preventDefault();
      const pag = parseInt(pagina, 10) > 1 ? Math.min(parseInt(pagina, 10), 1000) : 1;
      const q = document.getElementById('query-input').value.trim();
      const cat = document.getElementById('category-filter').value;
      const forceDui = document.getElementById('force-dui-check').checked;
      const resContainer = document.getElementById('results-container');
      const resumenEl = document.getElementById('search-resumen');
      const btn = document.getElementById('btn-search');
      const avisoEl = document.getElementById('search-aviso');
      const idSolicitud = ++gsSolicitudId;
      if (gsAbortar) { try { gsAbortar.abort(); } catch (_) { /* ya terminó */ } gsAbortar = null; }
      avisoEl.classList.add('hidden');
      resumenEl.classList.add('hidden');
      resumenEl.innerHTML = '';
      gsActivarAcciones(resContainer, function (n) { searchDiario(null, n > 0 ? n : gsPaginaActual); });

      const restaurarBoton = function () {
        btn.disabled = false;
        btn.innerHTML = '<i class="fa-solid fa-bolt"></i> Consultar Archivo Oficial';
        resContainer.removeAttribute('aria-busy');
      };

      if (q.length === 0) {
        restaurarBoton();
        avisoEl.innerText = 'Escribe un DUI, un nombre, un número de decreto o una frase entre comillas para buscar en el archivo.';
        avisoEl.classList.remove('hidden');
        resContainer.innerHTML = '';
        return;
      }
      gsPaginaActual = pag;
      if (!sinHistorial) gsActualizarUrl(q, cat, forceDui, pag, false);

      btn.disabled = true;
      btn.innerHTML = '<i class="fa-solid fa-spinner fa-spin"></i> Consultando Archivo...';
      resContainer.setAttribute('aria-busy', 'true');
      resContainer.innerHTML = '<div class="glass-card p-10 text-center text-gs-text2"><i class="fa-solid fa-spinner fa-spin text-3xl text-gs-accent"></i><p class="mt-3 text-sm font-mono">Buscando en los títulos, números de decreto, órganos emisores y en cada párrafo de las publicaciones oficiales...</p></div>';

      const url = '/api/v1/search?v=2&q=' + encodeURIComponent(q) +
                  (cat ? '&categoria=' + encodeURIComponent(cat) : '') +
                  (forceDui ? '&forceDui=1' : '') +
                  (pag > 1 ? '&pagina=' + pag : '');
      const control = typeof AbortController === 'function' ? new AbortController() : null;
      gsAbortar = control;

      fetch(url, control ? { signal: control.signal, headers: { Accept: 'application/json' } } : { headers: { Accept: 'application/json' } })
        .then(function (r) {
          return r.json().catch(function () { return {}; }).then(function (cuerpo) { return { status: r.status, ok: r.ok, data: cuerpo || {} }; });
        })
        .then(function (res) {
          if (idSolicitud !== gsSolicitudId) return; // llegó tarde: ya hay una consulta más reciente
          restaurarBoton();
          const data = res.data;
          if (res.status === 429) {
            gsMostrarFallo(resContainer, 'Demasiadas consultas seguidas', 'Espera unos segundos antes de volver a buscar.', true);
            return;
          }
          if (res.status === 400) {
            resContainer.innerHTML = '<div role="alert" class="glass-card p-6 rounded-xl text-center border border-gs-border"><i class="fa-solid fa-triangle-exclamation text-3xl text-gs-warning mb-2"></i><h4 class="text-base font-bold text-gs-text">Revisa lo que escribiste</h4><p class="text-xs sm:text-sm text-gs-text2 mt-1 max-w-md mx-auto">' + escapeHtmlText(data.aviso || 'La consulta no es válida.') + '</p></div>';
            return;
          }
          if (!res.ok || !Array.isArray(data.resultados)) {
            gsMostrarFallo(resContainer, 'El archivo no pudo responder', 'Hubo un problema en el servidor. Inténtalo de nuevo en unos segundos.', true);
            return;
          }
          if (data.aviso) {
            avisoEl.innerText = data.aviso;
            avisoEl.classList.remove('hidden');
          }
          // Entrada inválida sin resultados: nunca se dice «sin publicaciones». Con resultados (p. ej. DUI con verificador dudoso) se muestran junto al aviso.
          if (data.entradaInvalida && !(data.total > 0)) {
            resContainer.innerHTML = '<div role="alert" class="glass-card p-6 rounded-xl text-center border border-gs-border"><i class="fa-solid fa-triangle-exclamation text-3xl text-gs-warning mb-2"></i><h4 class="text-base font-bold text-gs-text">Revisa lo que escribiste</h4><p class="text-xs sm:text-sm text-gs-text2 mt-1 max-w-md mx-auto">' + escapeHtmlText(data.aviso || 'La consulta no es válida.') + ' Todavía no se hizo una consulta confiable, así que no podemos decir si aparece o no en el archivo.</p></div>';
            return;
          }

          gsConsultaActual = String(data.duiNormalized || q).replace(/["“”«»]/g, '');

          if (data.resultados.length === 0) {
            const fuera = data.total > 0 && pag > data.totalPaginas;
            resumenEl.innerHTML = '<p class="text-xs sm:text-sm text-gs-text">' + (fuera
              ? 'La página que pediste no existe: esta búsqueda solo tiene ' + escapeHtmlText(data.totalPaginas) + ' páginas de resultados.'
              : 'No se encontraron publicaciones que contengan todo lo que buscaste.') + '</p>';
            resumenEl.classList.remove('hidden');
            resContainer.innerHTML = '<div class="glass-card p-8 sm:p-10 rounded-xl text-center text-gs-text2 border border-gs-border">' +
              '<i class="fa-regular fa-circle-check text-4xl text-gs-accent mb-3"></i>' +
              '<h4 class="text-base sm:text-lg font-bold text-gs-text">Sin publicaciones oficiales encontradas</h4>' +
              '<p class="text-xs sm:text-sm text-gs-text2 mt-1 max-w-md mx-auto">El criterio «<strong>' + escapeHtmlText(q) + '</strong>» no aparece en los títulos, números de decreto, órganos emisores ni en el texto de las publicaciones de este tomo. Prueba con menos palabras, revisa la ortografía (no importan las tildes ni las mayúsculas) o quita el filtro de sección.</p>' +
              '<div class="mt-5"><button type="button" onclick="openModalAlert()" class="px-5 py-2.5 rounded-xl bg-gs-surface border border-gs-border text-gs-accent font-bold text-xs hover:bg-gs-surface2">Crear una alerta para este criterio</button></div></div>';
            return;
          }

          resumenEl.innerHTML = gsHtmlResumen(data);
          resumenEl.classList.remove('hidden');
          resContainer.innerHTML = gsRenderResultados(data);
          if (pag > 1) {
            resumenEl.setAttribute('tabindex', '-1');
            resumenEl.scrollIntoView({ behavior: 'smooth', block: 'start' });
            resumenEl.focus({ preventScroll: true });
          }
        })
        .catch(function (err) {
          if (idSolicitud !== gsSolicitudId) return;
          if (err && err.name === 'AbortError') return;
          restaurarBoton();
          gsMostrarFallo(resContainer, 'No pudimos conectar con el archivo oficial', 'Revisa tu conexión a internet y vuelve a intentarlo.', true);
        });
    }

    function showAlertMsg(ok, text) {
      const el = document.getElementById('alert-msg');
      el.innerText = text;
      el.className = 'text-xs rounded-xl px-3 py-2 ' + (ok ? 'bg-gs-surface2 text-gs-accent border border-gs-border' : 'bg-gs-surface2 text-gs-danger border border-gs-border');

    function submitAlert(e) {
      e.preventDefault();
      const crit = document.getElementById('alert-criterion').value.trim();
      const email = document.getElementById('alert-email').value.trim();
      const phone = document.getElementById('alert-phone').value.trim();

      const evaluado = evaluarCriterio(crit);
      if (!evaluado.ok) return showAlertMsg(false, evaluado.error);
      if (!document.getElementById('alert-consent').checked) return showAlertMsg(false, 'Para guardar la alerta debes aceptar el aviso de privacidad.');
      if (!/^[^@ ]+@[^@ ]+[.][^@ ]{2,}$/.test(email)) return showAlertMsg(false, 'Escribe un correo electrónico válido.');
      if (phone && !/^[+]?[0-9 ()-]{8,20}$/.test(phone)) return showAlertMsg(false, 'El teléfono solo admite números, espacios y el signo +.');

      fetch('/api/v1/alerts', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          criterioBusqueda: evaluado.valor,
          tipoCriterio: evaluado.tipo,
          usuarioEmail: email,
          telefonoWhatsapp: phone,
          consentimiento: true
        })
      })
      .then(async (r) => {
        const body = await r.json().catch(() => ({}));
        if (!r.ok) return showAlertMsg(false, body.error || 'No se pudo registrar la alerta. Revisa los datos.');
        showAlertMsg(true, 'Alerta registrada para «' + evaluado.valor + '».');
        setTimeout(closeModalAlert, 2500);
      })
      .catch(() => showAlertMsg(false, 'No pudimos conectar. Inténtalo de nuevo en unos segundos.'));
    }

        const t = ev.target && ev.target.closest ? ev.target : null;
        if (!t) return;
        const ab = t.closest('[data-ebook-pag]');
        if (ab) { openEbookViewer(parseInt(ab.getAttribute('data-ebook-pag'), 10) || 1, ab.getAttribute('data-term') || '', ab.getAttribute('data-dia') || undefined, ab.getAttribute('data-rec') || undefined); return; }
        const fb = t.closest('[data-focus-term]');
        if (fb) openEbookViewer(parseInt(fb.getAttribute('data-pag'), 10) || 1, fb.getAttribute('data-focus-term') || '', fb.getAttribute('data-dia') || undefined);
      });
      // La búsqueda se puede compartir por enlace (?q=...&categoria=...&forceDui=1&pagina=2) y el botón «atrás» la restaura.
      const aplicarUrl = (primera) => {
        let p;
        try { p = new URLSearchParams(location.search); } catch (_) { p = null; }
        const q = p ? (p.get('q') || '').slice(0, 200) : '';
        if (!q) { return; }
        document.getElementById('query-input').value = q;
        const cat = p.get('categoria') || '';
        const sel = document.getElementById('category-filter');
        sel.value = Array.from(sel.options).some((o) => o.value === cat) ? cat : '';
        document.getElementById('force-dui-check').checked = p.get('forceDui') === '1';
        handleDuiInputPreview(q);
        searchDiario(null, p.get('pagina'), true);
      };
      window.addEventListener('popstate', () => aplicarUrl(false));
      aplicarUrl(true);
    });
